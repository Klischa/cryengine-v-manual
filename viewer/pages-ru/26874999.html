Начальная настройка

Создайте новый проект из пустого template и откройте Game.cs. Game Класс является точкой входа приложения как в GameLauncher, так и в Sandbox. Когда приложение запускается, OnInitialize будет вызван метод. Это создаст новый экземпляр Game который можно использовать для начала настройки UI. Shutdown метод будет вызван при завершении воспроизведения в редакторе, который можно использовать для очистки UI.

private Canvas _canvas;
 
private Game()
{

  if(Engine.IsDedicatedServer)
  {
    return;
  }
  
  // Used to set up a new UI.
  CreateUI ();
  Mouse.ShowCursor();
}

Добавление холста и текста

Далее, Canvas будет добавлен в Application который нарисует Text компонент. Каждый UIElement и UIComponent должен быть ребенком Canvas который отвечает за обновление и управление каждым ребенком UIElement.

Canvas и Text добавляются в CreateUI добавив следующий код.

private void CreateUI()
{
  _canvas = SceneObject.Instantiate<Canvas>(null);
 
  var text = _canvas.AddComponent<Text>();
  text.Alignment = Alignment.Center;
  text.Height = 24;
  text.Content = "Basic UI Example";
  text.Offset = new Point(0f, 120f);
}

После компиляции проекта и запуска игры (в формате GameLauncher или Sandbox) текст появится чуть ниже центра экрана.

Добавление изображения

Далее будет добавлено изображение. Во-первых, щелкните правой кнопкой мыши и сохранить в твой 'Актив/libs/ui/папка. Из-за прозрачности изображение может выглядеть как пустое, но когда вы импортируете его в свой проект, вы сможете увидеть разницу.

Чтобы получить доступ к изображению в игре, нам нужно будет использовать ResourceManager и ImageFromFile метод. Это загружает ресурс по указанному пути в виде изображения.

Если изображение по указанному пути уже было загружено ранее, ResourceManager вернет кэшированную версию.

Развернуть CreateUI метод со следующим кодом:

private void CreateUI()
{
  // ...
  
  // Create a UIElement for our image.
  var logoElement = SceneObject.Instantiate<UIElement>(_canvas);
  logoElement.RectTransform.Size = new Point(300f, 300f);
  logoElement.RectTransform.Alignment = Alignment.Center;

  // Add an Image component for the image itself.
  var logoImage = logoElement.AddComponent<Image>();

  // Load our 'logo' image from 'Assets/libs/ui/'.
  logoImage.Source = ResourceManager.ImageFromFile(UIElement.DataDirectory + "/cryengine_logo.png", true);
}

ResourceManager является частью Resources namespace. Чтобы включить этот namespace в свой код, добавьте строку using CryEngine.Resources; в начале Game.cs файл.

Добавление кнопки

Последний компонент, который нужно добавить в этом уроке, — это кнопка, позволяющая выйти из игры при ее запуске. А Button это UIElement который имеет ButtonCtrl. Button отвечает за обработку визуального представления кнопки, а ButtonCtrl отвечает за обработку входных событий, состояние Button и содержимое кнопки. ButtonCtrl также содержит несколько обратных вызовов для обработки ввода.

  • OnPressed - Вызывается, когда кнопка нажата и снова отпущена без потери фокуса.
  • OnEnterMouse - Вызывается, когда мышь входит в прямоугольную область кнопки.
  • OnLeaveMouse - Вызывается, когда мышь покидает прямоугольную область кнопки.
  • OnFocusEnter - Вызывается, когда эта кнопка находится в фокусе, например, с помощью клавиатуры или контроллера.
  • OnFocusLost - Вызывается, когда эта кнопка теряет фокус.

Это обычные события C#, и к ним можно подключиться, как и к любому другому типу событий.

Чтобы создать экземпляр Button в нашем Canvas и назначьте обратный вызов OnPressed событие, необходимо добавить следующий код в CreateUI метод:

private void CreateUI()
{
  // ...

  // Create a quit button.
  var quitButton = SceneObject.Instantiate<Button>(_canvas);
  quitButton.RectTransform.Alignment = Alignment.Bottom;
  quitButton.RectTransform.Padding = new Padding(0f, -160f);
  quitButton.RectTransform.Size = new Point(300f, 30f);
  quitButton.Ctrl.Text.Content = "Quit";
  quitButton.Ctrl.Text.Height = 18;
  quitButton.Ctrl.Text.DropsShadow = true;
  quitButton.Ctrl.Text.Alignment = Alignment.Center;

  // Assign a callback to respond to the quit button being pressed.
  quitButton.Ctrl.OnPressed += OnQuitButtonPressed;
}

Наконец добавьте соответствующий OnQuitButtonPressed метод, который выйдет из игры.

private void OnQuitButtonPressed()
{
  Engine.Shutdown();
}

Перекомпилируйте и запустите игру. Теперь вы увидите Выйти кнопка, которая приведет к выходу из игры при нажатии.

Окончательный код

using System;
using CryEngine.UI;
using CryEngine.UI.Components;
using CryEngine.Resources;

namespace CryEngine.Game
{
  /// <summary>
  /// Basic sample of running a Mono Application.
  /// </summary>
  public class Game : IDisposable
  {
    private static Game _instance;

    private Canvas _canvas;
    
    private Game()
    {
      // The server doesn't support rendering UI and receiving input, so initializing those system is not required.
      if(Engine.IsDedicatedServer)
      {
        return;
      }
      CreateUI ();
      Mouse.ShowCursor();
    }

    public static void Initialize()
    {
      if(_instance == null)
      {
        _instance = new Game();
      }
    }

    public static void Shutdown()
    {
      _instance?.Dispose();
      _instance = null;
    }

    private void CreateUI()
    {
      _canvas = SceneObject.Instantiate<Canvas>(null);

      var text = _canvas.AddComponent<Text>();
      text.Alignment = Alignment.Center;
      text.Height = 24;
      text.Content = "Basic UI Example";
      text.Offset = new Point(0f, 120f);

      // Create a UIElement for our image.
      var logoElement = SceneObject.Instantiate<UIElement>(_canvas);
      logoElement.RectTransform.Size = new Point(300f, 300f);
      logoElement.RectTransform.Alignment = Alignment.Center;

      // Add an Image component for the image itself.
      var logoImage = logoElement.AddComponent<Image>();

      // Load our 'logo' image from 'Assets/libs/ui/'.
      logoImage.Source = ResourceManager.ImageFromFile(UIElement.DataDirectory + "/cryengine_logo.png", true);

      // Create a quit button.
      var quitButton = SceneObject.Instantiate<Button>(_canvas);
      quitButton.RectTransform.Alignment = Alignment.Bottom;
      quitButton.RectTransform.Padding = new Padding(0f, -160f);
      quitButton.RectTransform.Size = new Point(300f, 30f);
      quitButton.Ctrl.Text.Content = "Quit";
      quitButton.Ctrl.Text.Height = 18;
      quitButton.Ctrl.Text.DropsShadow = true;
      quitButton.Ctrl.Text.Alignment = Alignment.Center;
      
      // Assign a callback to respond to the quit button being pressed.
      quitButton.Ctrl.OnPressed += OnQuitButtonPressed;
    }

    private void OnQuitButtonPressed()
    {
      Engine.Shutdown();
    }

    public void Dispose()
    {
      if(Engine.IsDedicatedServer)
      {
        return;
      }
      
      _canvas.Destroy();
    }
  }
}
