Начальная настройка

Отправной точкой этого урока является MainMenu класс. Это будет корневой объект меню. Все остальные меню будут дочерними по отношению к главному меню.

using CryEngine;
using CryEngine.UI;

class MainMenu
{
  private Canvas _canvas;
  private MainMenuController _controller;
  private Menu _currentMenu;
  
  public bool Visible { get { return _canvas.Active; } set { _canvas.Active = value; } }
 
  public MainMenu(Canvas canvas)
  {
    _canvas = canvas;
    
    // Create the Controller that controls this menu.
    _controller = new MainMenuController(this);
    // Instantiate the menu.
    _currentMenu = new HomeMenu(canvas, _controller);
    
    Mouse.ShowCursor();
  }
}

Далее идет Menu класс и HomeMenu класс. HomeMenu в конечном итоге создаст UI и обработает взаимодействие пользователя с ним.

using CryEngine;
using CryEngine.UI;
 
abstract class Menu
{
  protected UIElement Root { get; private set; }
  protected UIElement View { get; private set; }
  protected MainMenuController Controller { get; private set; }
  
  protected Menu(UIElement root, MainMenuController controller)
  {
    Root = root;
    Controller = controller;
  }
}

class HomeMenu : Menu
{
  public HomeMenu(UIElement root, MainMenuController controller) : base(root, controller)
  {

  }
}

И, наконец, внутри Game класс экземпляра MainMenu и Canvas будет создан.

public class Game : IDisposable
{
  private static Game _instance;

  private Canvas _canvas;
  private MainMenu _mainMenu;
  
  private Game()
  {
    // The server doesn't support rendering UI and receiving input, so initializing those system is not required.
    if(Engine.IsDedicatedServer)
    {
      return;
    }

    _canvas = SceneObject.Instantiate<Canvas>(null);
        _mainMenu = new MainMenu(_canvas);
  }
  
  public static void Initialize()
  {
    if(_instance == null)
    {
      _instance = new Game();
    }
  }

  public static void Shutdown()
  {
    _instance?.Dispose();
    _instance = null;
  }
  
  public void Dispose()
  {
    if(Engine.IsDedicatedServer)
    {
      return;
    }
    
    _canvas.Destroy();
  }
}

Создание дома UI

Первое меню будет Главная меню, в котором будет Начать игру кнопка и Выйти на рабочий стол кнопка.

Для этой части логотип CRYENGINE должен находиться в каталоге ресурсов в /assets/libs/ui. Логотип можно скачать .

Главное меню будет управляться UIHomeMenu класс, который наследуется от UIElement класс. Класс отвечает за создание базового контейнера, в котором будут располагаться кнопки.

using System;
using CryEngine;
using CryEngine.Rendering;
using CryEngine.Resources;
using CryEngine.UI;

class UIHomeMenu : UIElement
{
  public override void OnAwake()
  {
    // Create a vertical panel strip.
    var panel = Instantiate<Panel>(this);
    panel.Background.Color = Color.Black.WithAlpha(0.5f);
    panel.RectTransform.Size = new Point(400f, Renderer.ScreenHeight);
    panel.RectTransform.Alignment = Alignment.Left;
    panel.RectTransform.Pivot = new Point(1f, 0.5f);
    panel.RectTransform.Padding = new Padding(50f, 0f);

    // Display the logo.
    var logo = Instantiate<Panel>(panel);

    // Load the logo from the 'libs/ui' folder.
    var path = System.IO.Path.Combine(DataDirectory, "logo.png");
    logo.Background.Source = ResourceManager.ImageFromFile(path, true);
    logo.RectTransform.Alignment = Alignment.Top;
    logo.RectTransform.Size = new Point(300f, 300f);
    logo.RectTransform.Padding = new Padding(0f, 150f); // Padding from top of the panel
  }
}

Обязательно включите CryEngine.Resources и CryEngine.Rendering namespace.

Вернувшись в HomeMenu class, меню должно быть создано и прикреплено к корню UIElement. Это делается с помощью нового метода в Menu класс, который создаст и правильно прикрепит UI к корню.

protected T CreateView<T>() where T : UIElement
{
  var view = SceneObject.Instantiate<T>(Root);
  view.RectTransform.Alignment = Alignment.Center;
  view.RectTransform.Size = new Point(Root.RectTransform.Width, Root.RectTransform.Height);
  
  // Store a reference to the View. This is needed later on.
  View = view;
  return view;
}

Метод будет вызываться изнутри HomeMenu класс для создания экземпляра UIHomeMenu.

private UIHomeMenu _view;

public HomeMenu(UIElement root, MainMenuController controller): base(root, controller)
{
  _view = CreateView<UIHomeMenu>();
}

Добавление кнопок

Теперь, когда базовый UI готов, Начать игру и Выйти на рабочий стол кнопки можно добавить.

Создание представления

Кнопки будут отображаться в столбце. UIButtonList будет выступать в качестве контейнера для кнопок меню. Чтобы было проще расположить кнопки, UIButtonList будет наследовать от VerticalLayoutGroup который специально создан для отображения элементов в столбце.

AddButton метод создаст экземпляр Button и добавит его в группу вертикального макета. Таким образом, кнопки можно легко добавить в меню.

using CryEngine;
using CryEngine.UI;
 
class UIButtonList : VerticalLayoutGroup
{
  public Button AddButton(string label)
  {
    // Make the button.
    var button = SceneObject.Instantiate<Button>(this);
    button.RectTransform.Size = new Point(RectTransform.Width, 40f);
    button.RectTransform.Alignment = Alignment.Top;
    button.RectTransform.Pivot = new Point(0.5f, 1f);
    button.Ctrl.Text.Content = label;

    // Add the button to the vertical layout group.
    Add(button);
    return button;
  }
}

Теперь, когда можно добавить кнопки UIButtonList готов к добавлению в UIHomeMenu. Это будет сделано в OnAwake метод UIHomeMenu. После создания экземпляра UIButtonList тот Начать игру и Выйти на рабочий стол к нему будут добавлены кнопки.

Для обработки нажатия кнопки необходимо добавить два новых события в UIHomeMenu также. Один StartPressed когда нажата кнопка запуска и один QuitToDesktopPressed когда нажата кнопка выхода. И, наконец, поле типа private UIButtonList названный _mainMenuButtons требуется сохранить ссылку на контейнер.

public event Action StartPressed;
public event Action QuitToDesktopPressed;

private UIButtonList _mainMenuButtons;


В конце OnAwake метод UIHomeMenu экземпляр UIButtonList будут созданы, а также добавлены кнопки и связанные с ними обратные вызовы.

// Create the UIButtonList.
_mainMenuButtons = SceneObject.Instantiate<UIButtonList>(panel);
_mainMenuButtons.RectTransform.Size = new Point(panel.RectTransform.Width - 50f, 400f);
_mainMenuButtons.RectTransform.Alignment = Alignment.Center;
 
// Assign callbacks to each button for when they are pressed.
_mainMenuButtons.AddButton("Start Game").Ctrl.OnPressed += () => StartPressed?.Invoke();
_mainMenuButtons.AddButton("Quit to Desktop").Ctrl.OnPressed += () => QuitToDesktopPressed?.Invoke();

Создание логики

Теперь, когда UI готов и доступны необходимые обратные вызовы, HomeMenu можно изменить для выполнения действия при нажатии кнопок.

Сначала связанный метод MainMenuController добавляются к событиям OnPressed UIHomeMenu.

class HomeMenu
{
  private UIHomeMenu _view;
  
  public HomeMenu(UIElement root, MainMenuController controller)
  {
    _view = CreateView<UIHomeMenu>();
    
    // Assign a callback to the button press event that will call the controller.
    _view.StartPressed += controller.StartGame;
    _view.QuitToDesktopPressed += controller.QuitToDesktop;
  }
}

Наконец, методы в MainMenuController реализованы так, что StartGame и QuitToDesktop выполнить действие. В MainMenuController class необходимо обновить следующие методы.

public void StartGame()
{
  Engine.Console.ExecuteString("map FirstMap", false, true);

  // Hide the menu.
  _mainMenu.Visible = false;
}

public void QuitToDesktop()
{
  Engine.Shutdown();
}

После компиляции и запуска игры меню станет функциональным. Когда Начать игру при нажатии будет загружена карта под названием FirstMap (если эта карта существует) и скройте UI. Если Выйти на рабочий стол при нажатии на движок отправляется сообщение о завершении работы, которое завершает игру.

Стилизация кнопок

В качестве завершающего штриха кнопки можно сделать более привлекательными.

Если навести указатель мыши на кнопку, должны произойти следующие действия:

  • отступ текста на несколько фреймов.
  • покажите изображение стрелки влево.
  • изменить цвет текста.

Изменение текста

Сначала появился новый класс под названием UIButton что происходит от Button требуется. OnAwake метод будет переопределен, поэтому он изменит стиль текста и отключит фоновое изображение.

using CryEngine;
using CryEngine.Resources;
using CryEngine.UI;
 
class UIButton : Button
{
  public override void OnAwake()
  {
    base.OnAwake();
    
    Background.Active = false; // Disable the background image.
    ShowPressedFrame = false; // Disable the frame the appears when pressing the button.

    Ctrl.Text.Alignment = Alignment.Left;
    Ctrl.Text.Height = 24;
    Ctrl.Text.DropsShadow = false;
  }
}

var button = SceneObject.Instantiate<Button>(this);

Перемещение текста

Далее текст должен сдвинуться вправо на определенное время, если на него наведена мышь, и вернуться в исходное положение, как только мышь уйдет.

Внутри UIButton class необходимо добавить следующие поля.

private bool _mouseOver;
private float _moveSpeed = 4.0f;
private float _moveProgress = 1.0f;
private float _targetOffset = 15.0f;

После этого SetOver, SetNormal и OnUpdate необходимо переопределить. Внутри этих методов цвета будут переопределены, а значение _mouseOver будет изменено. OnUpdate будет использоваться для плавного перемещения текста.

public override void SetOver()
{
  // Change the text color on mouse over.
  Ctrl.Text.Color = new Color(0.8f, 0.8f, 0.8f);
 
  if(!_mouseOver)
  {
    _mouseOver = true;
    _moveProgress = 1.0f - _moveProgress;
  }
}
public override void SetNormal()
{
  // Reset the text color when the mouse leaves the button.
  Ctrl.Text.Color = Color.White;
 
  if(_mouseOver)
  {
    _mouseOver = false;
    _moveProgress = 1.0f - _moveProgress;
  }
}
 
public override void OnUpdate()
{
  if(_moveProgress < 1.0)
  {
    _moveProgress = MathHelpers.Max(_moveProgress + _moveSpeed * FrameTime.Delta, 1.0f);
    var progress = _mouseOver ? _moveProgress : 1.0f - _moveProgress;
    var offset = MathHelpers.Lerp(0.0f, _targetOffset, progress);
    Ctrl.Text.Offset = new Point(offset, 0);
  }
}
