Низкоуровневая потоковая система

CryCommon Интерфейсы и структуры

Файл IStreamEngine.h в CryCommon содержит все важные интерфейсы и структуры, используемые остальной частью движка.

Прежде всего, это сам IStreamEngine. В приложении есть только один IStreamingEngine, и он управляет всеми возможными потоками ввода-вывода. Большая часть следующей информации взята непосредственно из документации внутри кода, поэтому всегда полезно прочитать сам код в файле IStreamEngine.h на предмет недостающей информации.

Самая важная функция в IStreamEngine — это функция StartRead, которая используется для запуска любой потоковые запросы.

UNIQUE_IFACE struct IStreamEngine
{
public:    
 
    // Description:
    //   Starts asynchronous read from the specified file (the file may be on a
    //   virtual file system, in pak or zip file or wherever).
    //   Reads the file contents into the given buffer, up to the given size.
    //   Upon success, calls success callback. If the file is truncated or for other
    //   reason can not be read, calls error callback. The callback can be NULL
    //   (in this case, the client should poll the returned IReadStream object;
    //   the returned object must be locked for that)
    // NOTE: the error/success/progress callbacks can also be called from INSIDE
    //   this function
    // Return Value:
    //   IReadStream is reference-counted and will be automatically deleted if
    //   you don't refer to it; if you don't store it immediately in an auto-pointer,
    //   it may be deleted as soon as on the next line of code,
    //   because the read operation may complete immediately inside StartRead()
    //   and the object is self-disposed as soon as the callback is called.
    virtual IReadStreamPtr StartRead (const EStreamTaskType tSource, const char* szFile, IStreamCallback* pCallback = NULL, StreamReadParams* pParams = NULL) = 0;
 
};

Вот поддерживаемые в настоящее время типы задач потоковой передачи, и этот enum следует расширить, если требуется потоковая передача нового типа объекта:

enum EStreamTaskType
{
    eStreamTaskTypeCount      = 13,
    eStreamTaskTypePak        = 12, // Pak file itself
    eStreamTaskTypeFlash      = 11, // Flash file object
    eStreamTaskTypeVideo      = 10, // Video data (when streamed)
    eStreamTaskTypeReadAhead  = 9,  // Read ahead data used for file reading prediction
    eStreamTaskTypeShader     = 8,  // Shader combination data
    eStreamTaskTypeSound      = 7, 
    eStreamTaskTypeMusic      = 6,
    eStreamTaskTypeFSBCache   = 5,  // Complete FSB file
    eStreamTaskTypeAnimation  = 4,  // All the possible animations types (dba, caf, ..)
    eStreamTaskTypeTerrain    = 3,  // Partial terrain data
    eStreamTaskTypeGeometry   = 2,  // Mesh or mesh lods
    eStreamTaskTypeTexture    = 1,  // Texture mip maps (currently mip0 is not streamed)
};

Объект обратного вызова может быть предоставлен функции StartStream, чтобы получать информацию о завершении запроса потоковой передачи. Объект обратного вызова должен реализовывать следующие 2 функции:

class IStreamCallback
{
public:
    // Description:
    //   Signals that reading the requested data has completed (with or without error).
    //   This callback is always called, whether an error occurs or not, and is called
    //   from the async callback thread of the streaming engine, which happens
    //   directly after the reading operation
    virtual void StreamAsyncOnComplete (IReadStream* pStream, unsigned nError) {}
 
    // Description:
    //   Same as the StreamAsyncOnComplete, but this function is called from the main
    //   thread and is always called after the StreamAsyncOnComplete function.
    virtual void StreamOnComplete (IReadStream* pStream, unsigned nError) = 0;
};

При запуске запроса на чтение также можно указать некоторые дополнительные необязательные параметры:

struct StreamReadParams
{
public:
     
    // The user data that'll be used to call the callback.
    DWORD_PTR dwUserData;
 
    // The priority of this read
    EStreamTaskPriority ePriority;
 
    // Description:
    //   The buffer into which to read the file or the file piece
    //   if this is NULL, the streaming engine will supply the buffer.
    // Notes:
    //   DO NOT USE THIS BUFFER during read operation! DO NOT READ from it, it can lead to memory corruption!
    void* pBuffer;
 
    // Description:
    //   Offset in the file to read; if this is not 0, then the file read
    //   occurs beginning with the specified offset in bytes.
    //   The callback interface receives the size of already read data as nSize
    //   and generally behaves as if the piece of file would be a file of its own.
    unsigned nOffset;
 
    // Description:
    //   Number of bytes to read; if this is 0, then the whole file is read,
    //   if nSize == 0 && nOffset != 0, then the file from the offset to the end is read.
    //   If nSize != 0, then the file piece from nOffset is read, at most nSize bytes
    //   (if less, an error is reported). So, from nOffset byte to nOffset + nSize - 1 byte in the file.
    unsigned nSize;
 
    // Description:
    //   The combination of one or several flags from the stream engine general purpose flags.
    // See also:
    //   IStreamEngine::EFlags
    unsigned nFlags;
};

Возвращаемое значение функции StartRead — это объект IReadStream, который при необходимости может быть сохранен на клиенте. Объект IReadStream пересчитывается внутренне. Если объект обратного вызова может быть уничтожен до завершения операции чтения, тогда поток чтения должен где-то храниться, и для него необходимо вызвать прерывание. Это позаботится о внутренней очистке всего запроса на чтение, а также вызовет функции обратного вызова async и sync.

Функцию ожидания можно использовать для выполнения блокирующих запросов на чтение в механизме потоковой передачи. Эту функцию можно использовать из потока асинхронного чтения, который хочет использовать систему потоковой передачи Cryengine для выполнения фактического чтения.

class IReadStream : public CMultiThreadRefCount
{
public:
    virtual void Abort() = 0;
    virtual void Wait( int nMaxWaitMillis=-1 ) = 0;
};

Внутренний поток запроса на чтение

CryENGINE StreamEngine использует дополнительный рабочий процесс и IO-Threads внутри. Для каждого возможного входа IO создается отдельный StreamingIOThread, который может работать независимо друг от друга.

На данный момент у него есть следующие темы IO:

  • Оптический: потоковая передача с оптического накопителя данных.
  • HDD : потоковая передача данных, установленных на HDD (это могут быть полностью установленные данные игры или данные теневого копирования).
  • Память: потоковая передача из упакованных в память файлов, для которой вообще не требуется никакого реального IO.

Когда запрос на чтение делается в механизме потоковой передачи, он сначала проверяет, какой поток IO использовать, и вычисляет ключ сортировки. Затем запрос вставляется в один из IOThreads.

После завершения операции чтения запрос перенаправляется в один из потоков распаковки (если данные были сжаты), а затем в один из потоков асинхронного обратного вызова. Количество потоков асинхронного обратного вызова зависит от платформы, а некоторые потоки асинхронного обратного вызова зарезервированы для определенных типов потоковых запросов, таких как геометрия и текстуры. После обработки асинхронного обратного вызова готовый запрос потоковой передачи добавляется в механизм потоковой передачи для обработки в основном потоке. Следующее обновление механизма потоковой передачи из основного потока вызовет обратный вызов синхронизации (StreamOnComplete) и при необходимости очистит временно выделенную память.

Для получения информации о IO/WorkerThreads проверьте классы StreamingIOThread и StreamingWorkerThread.

Сортировка запросов на чтение

Запросы к потоковому движку нет обрабатываются в том же порядке, в котором они были запрошены. Система пытается внутренне «оптимизировать» порядок чтения данных, чтобы максимизировать пропускную способность чтения.

При чтении данных с оптического диска очень важно уменьшить количество переборов (и с HDD тоже, но в меньшей степени). Один поиск может занять более 100 мс, тогда как фактическое время чтения составит всего несколько мс. Немного официальной статистики 360 XDK:

  • Пропускная способность внешнего диаметра: 12× (приблизительно 15 MB в секунду).
  • Пропускная способность внутреннего диаметра: 5× (6,8 MB в секунду).
  • Среднее время поиска (1/3 хода): типичное значение 110 мс, максимальное 140 мс.
  • Время поиска при полном ходе: типичное значение 180 мс, максимальное 240 мс.
  • Время переключения слоев: 75 мс.

Алгоритм внутренней сортировки учитывает следующие правила в следующем порядке:

  • приоритет запроса. Запросы с высоким приоритетом всегда имеют приоритет, но могут быть очень дорогими, если их много (может быть введено много дополнительных запросов).
  • группировка по времени - Запросы, сделанные в течение определенного времени, группируются для создания непрерывной операции чтения на диске для каждой временной группы. Значение по умолчанию — 2 секунды, но его можно изменить с помощью следующей переменной: sys_streaming_requests_grouping_time_period. Группировка по времени оказывает огромное влияние на среднее время выполнения запросов. Это увеличивает время выполнения некоторых запросов на быстрое чтение, но радикально сокращает общее время завершения, поскольку большая часть запросов потоковой передачи поступает из «случайных» мест на диске.
  • фактический смещение на диске - Фактическое смещение диска рассчитывается и используется во время сортировки. Файлы с более высоким смещением получают более высокий приоритет, поэтому важно организовать структуру диска так, чтобы она отражала желаемый порядок потоковой передачи.

Для Crysis 2 использовался следующий порядок: звуки/музыка, анимация, сетки, текстуры с низким разрешением, текстуры с высоким разрешением.

Для получения информации о сортировке обратитесь к исходному коду в StreamAsyncFileRequest::ComputeSortKey(). Вот основной код сортировки:

void CAsyncIOFileRequest::ComputeSortKey(uint64 nCurrentKeyInProgress)
{
    .. compute the disc offset (can be requested using CryPak)
 
    // group items by priority, then by snapped request time, then sort by disk offset
    m_nDiskOffset += m_nRequestedOffset;
    m_nTimeGroup = (uint64)(gEnv->pTimer->GetCurrTime() / max(1, g_cvars.sys_streaming_requests_grouping_time_period));
    uint64 nPrioriry = m_ePriority;
 
    int64 nDiskOffsetKB = m_nDiskOffset >> 10; // KB
    m_nSortKey = (nDiskOffsetKB) | (((uint64)m_nTimeGroup) << 30) | (nPrioriry << 60);
}

Потоковая статистика

Механизм потоковой передачи можно запросить для получения статистики потоковой передачи с помощью функции GetStreamingStatistics().

Большая часть статистики разделена на 2 группы: одна собирается в последнюю секунду, а другая — из последней (что обычно происходит во время загрузки уровня), но ее также можно принудительно сбросить в игре.

Информация о типе носителя предоставляет информацию для каждой системы ввода IO (HDD, оптическая, память).

struct SMediaTypeInfo
{
    // stats collected during the last second
    float fActiveDuringLastSecond;
    float fAverageActiveTime;  
    uint32 nBytesRead;     
    uint32 nRequestCount;
    uint64 nSeekOffsetLastSecond;
    uint32 nCurrentReadBandwidth;
    uint32 nActualReadBandwidth;    // only taking actual reading time into account
 
    // stats collected since last reset    
    uint64 nTotalBytesRead;    
    uint32 nTotalRequestCount; 
    uint64 nAverageSeekOffset;  
    uint32 nSessionReadBandwidth;      
    uint32 nAverageActualReadBandwidth; // only taking actual read time into account
};

Информация о типе запроса предоставляет информацию о каждом типе запроса потоковой передачи (геометрия, текстуры, анимация и т. д.).

struct SRequestTypeInfo
{
    int nOpenRequestCount; 
    int nPendingReadBytes; 
 
    // stats collected during the last second
    uint32 nCurrentReadBandwidth;  
     
    // stats collected since last reset
    uint32 nTotalStreamingRequestCount;
    uint64 nTotalReadBytes;     // compressed data
    uint64 nTotalRequestDataSize;   // uncompressed data
    uint32 nTotalRequestCount;
    uint32 nSessionReadBandwidth;  
 
    float fAverageCompletionTime;   // Average time it takes to fully complete a request
    float fAverageRequestCount; // Average amount of requests made per second
};

Глобальная статистика, содержащая все объединенные данные:

struct SStatistics

    uint32 nTotalCurrentReadBandwidth;  // Total bytes/sec over all types and systems.

    float fAverageCompletionTime;   // Time in seconds on average takes to complete read request.
    float fAverageRequestCount; // Average requests per second being done to streaming engine

Потоковая передача и файлы levelcache pak

Как упоминалось ранее, очень важно минимизировать поиск и расстояние поиска при чтении с оптического носителя, поэтому система сборки немного модифицирована для оптимизации внутреннего расположения данных для потоковой передачи.

Самый простой и быстрый результат — вообще не делать никаких IO, а читать данные из сжатых данных в памяти. Для этого создаются небольшие паки для запуска и по уровням. Они загружаются во время загрузки уровня в память, а некоторые остаются там до конца уровня. Остальные используются только для ускорения загрузки уровня. Все небольшие файлы и небольшие запросы на чтение в идеале должны перенаправляться в эти пакеты.

Для создания этих пакетов используется специальный файл сборки RC_Job: Bin32/rc/RCJob_PerLevelCache.xml. Эти пакеты генерируются во время обычного конвейера сборки. Внутреннее управление движком осуществляется классом CResourceManager, который использует глобальный SystemEvents для предварительной загрузки или выгрузки пакетов.

В настоящее время во время загрузки уровня в память загружаются следующие пакеты (sys_PakLoadCache):

  • level.pak - Содержит все фактические данные уровня, и после загрузки уровня к ним больше нельзя прикасаться.
  • xml.pak
  • dds0.pak - Содержит все самые низкие MIP всех текстур на уровне.
  • cgf.pak & cga.pak - Загрузка только при включенной потоковой передаче CGF.

Эти пакеты кэшируются в памяти во время процесса загрузки уровня (sys_PakStreamCache):

  • dds_cache.pak - Содержит все файлы dds размером менее 6 KB (кроме файлов dds.0).
  • cgf_cache.pak - Содержит все файлы cgf размером менее 32 KB (только если включена потоковая передача CGF).

Без этих пакетов загрузка уровня может занять до нескольких минут, а производительность потоковой передачи значительно снижается, поэтому убедитесь, что они доступны!

Информация обо всех ресурсах уровня хранится в resourcelist.txt и auto_resourcelist.txt. Эти файлы генерируются системой автоматического тестирования, которая загружает каждый уровень и выполняет на нем предварительно записанное прохождение. Эти файлы списка ресурсов используются на этапе сборки для создания пакетов уровней.

Все данные, которых нет в этих пакетах памяти, фактически выполняются IO на оптическом приводе или HDD, и здесь также лучше уменьшить количество операций поиска. Этот этап оптимизации также выполняется в процессе сборки с использованием RC.

Все данные, которые можно передавать в потоковом режиме, извлекаются из всех списков ресурсов всех уровней, удаляются из файлов пакетов по умолчанию (Objects.pak, Textures.pak, Animations.pak, ...) и помещаются в новые оптимизированные пакеты для потоковой передачи внутри папки потоковой передачи.

При создании потоковых пакетов используются следующие правила:

  • Разделить по расширение: разные файлы расширений помещаются в разные пакеты (dds, caf, dba, cgf, ...), чтобы файлы одного типа располагались близко друг к другу, чтобы их можно было читать пакетами. Эти пакеты также используются для повышения приоритета определенных типов файлов во время сортировки запросов с использованием фактического смещения диска.
  • Разделить на DDS тип: разные типы dds сортируются по-разному, чтобы увеличить приоритет разных типов (например, диффузные карты имеют более высокий приоритет, чем обычные карты). Фактическое расстояние в пакете используется при сортировке запроса.
  • Разделить на DDS мип: Самые высокие мипы вынесены в отдельный пак-файл. Обычно они занимают более 60% размера всех меньших MIP-файлов и затем могут транслироваться с более низким приоритетом, а среднее время поиска для чтения меньших текстур значительно сокращается. Система потоковой передачи текстур внутренне оптимизирует чтение, чтобы отразить эти разделенные данные текстуры.
  • Сортировать По алфавиту: Сортировка по алфавиту необходима, поскольку некоторые данные (например, CGF во время загрузки уровня MP) загружаются в алфавитном порядке, и изменение этого правила может серьезно повлиять на время загрузки!

Фактический код сортировки жестко запрограммирован в RC и его можно найти по адресу: Code\Tools\RC\ResourceCompiler\PakHelpers.cpp.

Очень важно, чтобы любые изменения оператора сортировки в RC также отражались внутри операторов сортировки механизма потоковой передачи текстур и потоковой передачи!

Однопоточный доступ к IO и неверный доступ к файлу

Очень важно, чтобы только один поток имел доступ к устройствам IO одновременно. Если несколько потоков одновременно читают с одного и того же устройства IO, то скорость чтения снижается более чем вдвое, и на чтение всего нескольких KB может потребоваться несколько секунд!

Причина в том, что считывающая головка IO частично прочитает несколько KB для одного потока, а затем прочитает еще несколько KB для другого потока, всегда выполняя при этом дорогостоящий поиск между ними.

Решение состоит в том, чтобы читать исключительно из StreamingIOThreads во время игры. CryENGINE по умолчанию отображает «Неверный доступ к файлу» предупреждение в верхнем левом углу при чтении данных не из того потока и остановке на 3 секунды, чтобы имитировать фактическую остановку при чтении с оптического привода.

Для получения дополнительной информации о том, как отслеживать и исправлять «Неверный доступ к файлам» в игре, посетите следующую страницу слияния: http://confluence/display/RND/Tracking+File+Access

Использование механизма потоковой передачи высокого уровня

Расширить текущую функциональность потоковой передачи очень легко с помощью StreamEngine. Ниже описан небольшой пример класса о том, как добавить новый тип потоковой передачи.

Создайте класс, производный от IStreamCallback (используется для информирования о завершении потоковой передачи), и добавьте некоторые базовые функции для чтения файла. Файл можно прочитать напрямую или с помощью механизма потоковой передачи. Когда данные считываются напрямую, он вызывает функцию ProcessData для фактического анализа загруженных данных. Функция также вызывается из асинхронного обратного вызова. При необходимости здесь можно выполнить некоторую обработку данных, поскольку она не выполняется в основном потоке.

Параметры по умолчанию используются при запуске запроса на чтение в механизме потоковой передачи. Также возможно предусмотреть само конечное хранилище данных. Это помогает уменьшить объем динамических выделений, выполняемых механизмом потоковой передачи.

В классе также хранится объект потока чтения, чтобы получить информацию о запросе потоковой передачи или иметь возможность отменить его при уничтожении объекта обратного вызова. Указатель сбрасывается в обратном вызове синхронизации, поскольку после этого вызова механизм потоковой передачи больше не будет ссылаться на него.

#include
class CNewStreamingType : public IStreamCallback
{
public:
    CNewStreamingType() : m_pReadStream(0), m_bIsLoaded(false) {}
    ~CNewStreamingType()
    {
        if (m_pReadStream)
            m_pReadStream->Abort();
    }
 
    // Start reading some data
    bool ReadFile(const char* acFilename, bool bUseStreamingEngine)
    {
        if (bUseStreamingEngine)
        {
            StreamReadParams params;
            params.dwUserData = eLoadFullData;
            params.ePriority = estpNormal;
            params.nSize = 0; // read the full file
            params.pBuffer = NULL; // don't provide any buffer, but copy data when streaming is done
 
            m_pReadStream = g_pISystem->GetStreamEngine()->StartRead(eStreamTaskTypeNewType, acFilename, this, &params);
        }
        else
        {
            // old way of reading file in a sync way (blocking call!)
            const char* acData = 0;
            size_t stSize = 0;
 
            .. read file directly using CryPak or fopen/fread
 
            ProcessData(acData, stSize);
            m_bIsLoaded = true;        
        }
 
        return m_bIsLoaded;
    }
 
    // Check if the data is ready and loaded
    bool IsLoaded() const { return m_bIsLoaded; }
 
protected:
 
    // implement the IStreamCallback function
    void StreamAsyncOnComplete(IReadStream* pStream, unsigned nError)
    {
        if(nError)
        {
            return;
        }
 
        const char* acData = (char*)pStream->GetBuffer();
        size_t stSize= pStream->GetBytesRead();
 
        ProcessData(acData, stSize);
        m_bIsLoaded = true;
    }
 
    void StreamOnComplete (IReadStream* pStream, unsigned nError)
    {
        m_pReadStream = 0;
    }  
 
    // process the actual loaded data
    void ProcessData(const char* acData, size_t stSize);
 
    // store the stream callback object to be sure it can be canceled when the object is destroyed
    IReadStreamPtr m_pReadStream;
    // Extra flag used to check if the data is ready
    bool m_bIsLoaded;
}
