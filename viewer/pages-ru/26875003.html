using System;
using CryEngine.Rendering;
using CryEngine.UI;
 
namespace CryEngine.Game
{
  public class Game
  {
    private static Game _instance;
 
    private Canvas _canvas;
    private PopupFactory _popupFactory;
   
    private Game()
    {
      // The server doesn't support rendering UI and receiving input, so initializing those system is not required.
      if(Engine.IsDedicatedServer)
      {
        return;
      }
 
      //Only move to the map if we're not in the sandbox. The sandbox can open the map all by itself.
      if(!Engine.IsSandbox)
      {
        Engine.Console.ExecuteString("map example", false, true);
      }
    }
 
    public static void OnLevelLoaded()
    {
      _instance?.CreateUI();
    }
 
    private void CreateUI()
    {
      Mouse.ShowCursor();
      _canvas = SceneObject.Instantiate<Canvas>(null);
      _canvas.RectTransform.Size = new Point(Renderer.ScreenWidth, Renderer.ScreenHeight);
      _popupFactory = new PopupFactory(_canvas);
    }
 
    public static void Initialize()
    {
      if(_instance == null)
      {
        _instance = new Game();
      }
    }
 
    public static void Shutdown()
    {
      _instance._canvas.Destroy();
      _instance = null;
    }
  }
}

    private readonly Color MediumBlue = new Color(0.603f, 0.708f, 0.822f);
        private readonly Color LightBlue = new Color(0.573f, 0.678f, 0.792f);

        public override void OnAwake()
        {
            base.OnAwake();

            // Set the background color of the panel.
            Background.Color = DarkBlue;
 
      // Add a frame around the outside.
      var frame = AddComponent<Image>();

      frame.Source = ResourceManager.ImageFromFile(path);
      frame.SliceType = SliceType.Nine;

            // Set the size of the popup.

Тип фрагмента изображения имеет три возможных значения: ThreeHorizontal, ThreeVerticalи Nine. На диаграмме ниже показано, как каждый тип фрагментов разбивает изображение и как эти фрагменты растягиваются.

Добавление кнопок

Во всплывающем окне по-прежнему отсутствует способ взаимодействия пользователя с ними. Для получения пользовательских кнопок ввода будут добавлены во всплывающее окно. Кнопки будут добавлены в группу горизонтального макета и будут расположены автоматически. Будут добавлены дополнительные функции для автоматической регулировки ширины всех существующих кнопок для равномерного распределения при добавлении новой кнопки.

Внутри UIPopup следующее будет добавлено в начало класса.

public event Action<UIPopup> Shown;
public event Action<UIPopup> Closed;
 
private HorizontalLayoutGroup _buttonGroup;
private  List<Button> _buttons;

В конце OnAwake метод, для создания HorizontalLayoutGroup для кнопок и для инициализации List<Button>.

_buttonGroup = Instantiate<HorizontalLayoutGroup>(_innerContainer);
_buttonGroup.RectTransform.Size = new Point(_innerContainer.RectTransform.Width, 20f);
_buttonGroup.RectTransform.Alignment = Alignment.Bottom;
_buttonGroup.RectTransform.Pivot = new Point(0.5f, 0f);
 
_buttons = new List<Button>();

Добавлен следующий метод, который будет обрабатывать создание экземпляров кнопок. Размер кнопок во всплывающем окне автоматически изменяется в соответствии со всеми добавленными кнопками. Ширина каждой кнопки устанавливается равной ширине внутреннего контейнера всплывающего окна, разделенной на количество кнопок, добавленных во всплывающее окно.

public void AddButton(string label, Action onPress)
{
  // Instantiate a button
  var button = Instantiate<Button>(_buttonGroup);
  button.RectTransform.Size = new Point(_buttonGroup.RectTransform.Width, 30f);
  button.RectTransform.Alignment = Alignment.Left;
  button.RectTransform.Pivot = new Point(1f, 0.5f);
  button.Ctrl.Text.Content = label;
  button.Ctrl.Text.Height = 18;
  button.Ctrl.Text.FontStyle = System.Drawing.FontStyle.Bold;
  button.Ctrl.Text.DropsShadow = false;
  button.Background.Color = MediumBlue;
  
  // Store a reference to the button.
  _buttons.Add(button);

 
  // Set width based on number of buttons.
  var width = _innerContainer.RectTransform.Width;
  _buttons.ForEach(x => x.RectTransform.Width = width / _buttons.Count);
 
  // Add the button to the horizontal group.
  _buttonGroup.Add(button);
}


Последнее, что необходимо назначить, это onPress обратный вызов, который добавляется к кнопке OnPressed событие. Это достигается добавлением следующего кода в конец AddButton метод.

button.Ctrl.OnPressed += () =>
{
  Closed?.Invoke(this);
  onPress?.Invoke();
};

Реализация всплывающего окна с информацией

Теперь, когда всплывающее окно UI готово, можно создать функциональное всплывающее окно с информацией.

Система всплывающих окон позволяет одновременно отображать только одно всплывающее окно. Если одновременно отображаются другие всплывающие окна, их необходимо скрыть, а также пометить соответствующим образом, чтобы при закрытии текущего всплывающего окна отображалось следующее всплывающее окно, ожидающее отображения.

Для начала необходимо внести некоторые изменения в UIPopup.

В верхней части класса следующее свойство добавлен.

public bool FlaggedForShow { get; private set; }

Далее следующие два метода, которые вызовут Shown и Closed события добавляются.

public void Show()
{
  FlaggedForShow = true;
  Shown?.Invoke(this);
}
 
public void Close()
{
  FlaggedForShow = false;
  Closed?.Invoke(this);
}

Внутри PopupFactory, Shown и Closed события подключаются для отображения и уничтожения всплывающего окна.

Также необходимо отслеживать, когда всплывающее окно добавляется и удаляется. Таким образом, когда всплывающее окно будет закрыто, будут показаны все оставшиеся всплывающие окна, которые есть в системе.

Первым в классе А List<UIPopup> создан.

private List<UIPopup> _popups = new List<UIPopup>();

MakePopup Функциональность метода также расширена за счет добавления обратных вызовов в Shown и Closed события.

private UIPopup MakePopup(string header, string content)
{
    var popup = SceneObject.Instantiate<UIPopup>(_canvas);
    popup.Header = header;
    popup.Content = content;

    // Register popup.
    popup.Shown += OnShowPopup;
    popup.Closed += OnClosePopup;
 
  _popups.Add(popup);
 
  popup.Active = false;

    return popup;
}

И добавляются связанные обработчики.

private void OnShowPopup(UIPopup popup)
{
  Mouse.ShowCursor();

  // Display popup then bring it to front.
  popup.Active = true;
  popup.SetAsLastSibling();
}
 
private void OnClosePopup(UIPopup popup)
{
  Mouse.HideCursor();
 
  // Unregister popup.
  popup.Shown -= OnShowPopup;
  popup.Close -= OnClosePopup;
  _popups.Remove(popup);
 
  popup.Destroy();
 
  // Find a popup that is waiting to be shown.
  var nextPopup = _popups.FirstOrDefault(x => x.FlaggedForShow == true);
  if (nextPopup != null)
  {
      // Show the next popup waiting to be shown.
       nextPopup.Show();
  }
  else
  {
    // If there's no popup to show the mouse can be hidden.
       Mouse.HideCursor();
  }
}

И добавлен новый метод, который создаст всплывающее информационное окно.

public UIPopup MakeInfoPopup(string content, Action onOkay = null)
{
  var popup = MakePopup("Confirm", content);
  popup.AddButton("Okay", onOkay);
  return popup;
}

Вернувшись в Game класс, MakeTestPopup строку необходимо заменить на следующую.

Обратите внимание, что теперь нам нужно явно вызвать Show для отображения всплывающего окна.

Добавление фоновой панели

Теперь, когда базовая система создана, следующим шагом будет отображение прозрачной черной панели позади всплывающего окна, когда оно отображается, чтобы помочь отличить всплывающее окно от основного UI.

Следующее поле добавляется вверху страницы. PopupFactory класс.

private Panel _panel;

Затем в конструкторе PopupFactory а Panel создается с черным фоном.

public PopupFactory(Canvas canvas)
{
  _canvas = canvas;
  _popups = new List<UIPopup>();
  _panel = SceneObject.Instantiate<Panel>(_canvas);
  _panel.Background.Color = Color.Black.WithAlpha(0.8f);
  _panel.RectTransform.Size = new Point(Renderer.ScreenWidth, Renderer.ScreenHeight);
  _panel.RectTransform.Alignment = Alignment.Center;

Поскольку панель поначалу неактивна, необходима некоторая логика, которая ее активирует. Active состояние панели будет меняться, когда всплывающее окно отображается и закрывается.

OnShowPopup метод модифицирован для достижения этой цели.

private void ShowPopup(UIPopup popup)
{
  Mouse.ShowCursor();

  // Enable then bring the panel to front.
  _panel.Active = true;
  _panel.SetAsLastSibling();

  // Display popup then bring it to front.
  popup.Active = true;
  popup.SetAsLastSibling();
}

И OnClosePopup метод изменен для повторной деактивации панели.

// Find a popup that is waiting to be shown.
var nextPopup = _popups.FirstOrDefault(x => x.FlaggedForShow == true);
if (nextPopup != null)
{
   // Show the next popup waiting to be shown.
   ShowPopup(nextPopup);
}
else
{
   Mouse.HideCursor();
  _panel.Active = false;
}

После компиляции и запуска игры всплывающее окно теперь отображается с фоновой панелью позади него. Когда всплывающее окно закроется, фоновая панель также исчезнет.

Реализация дополнительных всплывающих окон

Имея основу, можно легко реализовать дополнительные всплывающие окна. В PopupFactory можно добавить новый метод, который будет создавать всплывающее окно в стиле Да/Нет.

public UIPopup MakeConfirmationPopup(string content, Action onYes, Action onNo = null)
{

  popup.AddButton("Yes", onYes);
  popup.AddButton("No", onNo);
  return popup;
}

Или можно добавить всплывающее окно в стиле Да/Нет/Отмена.

public UIPopup MakeConfirmationCancelPopup(string content, Action onYes, Action onNo, Action onCancel)
{
  var popup = MakePopup("Confirm", content);
  popup.AddButton("Yes", onYes);
  popup.AddButton("No", onNo);
  popup.AddButton("Cancel", onCancel);
  return popup;
}
